# GIS 矢量数据 Agent —— 依赖清单
#
# 安装：
#   python -m venv .venv
#   .venv\Scripts\python.exe -m pip install -r requirements.txt      # Windows
#   .venv/bin/python -m pip install -r requirements.txt              # macOS / Linux
#
# 实测环境：Windows 11 + Python 3.13.14
#   GeoPandas 1.2.0 / pyogrio 0.13.0 / Shapely 2.2.0 / pyproj 3.8.0
#   pandas 3.1.0 / NumPy 2.5.4 / PySide6 6.12.0
# Python 3.10 ~ 3.13 都有对应 wheel。

# ---------------------------------------------------------------------------
# 核心：矢量数据读写 + 空间运算
# ---------------------------------------------------------------------------
geopandas>=1.0,<2
pyogrio>=0.10            # GeoPandas 1.x 的默认 IO 引擎，自带 GDAL，无需另装
shapely>=2.0,<3          # 2.x 的向量化 API（构建/简化/拾取全程依赖）
pyproj>=3.6              # 坐标系转换（距离、面积统一投到 CGCS2000 3 度带）
pandas>=2.2,<4
numpy>=1.26,<3
# 矢量流式渲染的核心：pyogrio.open_arrow 的 RecordBatch 由 pyarrow 消费。
# 500 万要素不再整窗口物化成 GeoDataFrame，渲染线程按批解码 WKB。
pyarrow>=15

# Esri FileGDB（.gdb 文件夹数据库）：纯 Python 实现，不依赖 GDAL / fiona。
# 如果暂时不打开 GDB，也可以先不装；打开 .gdb 时会给出安装提示。
pyopenfilegdb @ git+https://github.com/lizhiziwang/pyopengdb.git

# ---------------------------------------------------------------------------
# 桌面界面：纯 Qt 原生控件 + QPainter 自绘地图（无 WebView）
# ---------------------------------------------------------------------------
PySide6>=6.5,<7          # 会一并装上 PySide6-Essentials / PySide6-Addons / shiboken6

# ---------------------------------------------------------------------------
# 栅格数据（GeoTIFF 等）：渲染 + 通用栅格算子（裁剪/重投影/重采样/计算/统计…）
# 没装也能启动，只是打开 .tif 或执行栅格算子时会提示缺这个包。
# ---------------------------------------------------------------------------
rasterio>=1.4            # 自带 GDAL，按视口抽取像素；依赖 affine / cligj / attrs / pyparsing

# ---------------------------------------------------------------------------
# 可选补充
# ---------------------------------------------------------------------------
# fiona>=1.9             # pyogrio 读不了的老格式兜底（app/io_utils.py 会自动尝试）

# ---------------------------------------------------------------------------
# 数据库连接：左栏「数据库」页签（PostgreSQL / PostGIS）
# 没装也能启动，点「测试连接 / 浏览表」时才提示安装。自带的 GDAL（pyogrio /
# rasterio 的 wheel）没编 PostgreSQL 驱动，所以直连走 psycopg2，几何用 ST_AsBinary。
# ---------------------------------------------------------------------------
